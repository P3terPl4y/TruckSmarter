from django.contrib.auth import get_user_model

User = get_user_model()


class VehicleModelTest(TestCase):
    """Pruebas para los modelos de vehículos."""

    def setUp(self):
        self.brand = Brand.objects.create(name="Toyota")
        self.user = User.objects.create_user(
            username="seller",
            email="seller@test.com",
            password="testpass123",
            role="admin",
        )
        self.vehicle = Vehicle.objects.create(
            brand=self.brand,
            model="Corolla",
            year=2020,
            price=20000.00,
            mileage=15000,
            transmission="automatic",
            fuel_type="gasoline",
            status="used",
            description="Excelente estado",
            created_by=self.user,
            is_active=True,
            sold=False,
        )

    def test_vehicle_creation(self):
        """Verifica que un vehículo se cree correctamente."""
        self.assertEqual(self.vehicle.brand.name, "Toyota")
        self.assertEqual(self.vehicle.model, "Corolla")
        self.assertEqual(str(self.vehicle), "Toyota Corolla (2020)")

    def test_mark_as_sold(self):
        """Verifica que mark_as_sold funcione correctamente."""
        from sales.models import Sale

        buyer = User.objects.create_user(
            username="buyer",
            email="buyer@test.com",
            password="testpass123",
            role="customer",
        )

        sale = self.vehicle.mark_as_sold(buyer)

        self.vehicle.refresh_from_db()
        self.assertTrue(self.vehicle.sold)
        self.assertFalse(self.vehicle.is_active)
        self.assertIsNotNone(sale)
        self.assertEqual(sale.buyer, buyer)
        self.assertEqual(sale.seller, self.user)
        self.assertEqual(sale.amount, self.vehicle.price)


class VehicleListViewTest(TestCase):
    """Pruebas para VehicleListView."""

    def setUp(self):

        self.user = User.objects.create_user(
            username="seller",
            email="seller@test.com",

            role="admin",
        )
        self.vehicle = Vehicle.objects.create(
            brand=self.brand,
            model="Corolla",
            year=2020,
            price=20000.00,
            mileage=15000,
            transmission="automatic",
            fuel_type="gasoline",
            status="used",
            created_by=self.user,
            is_active=True,
            sold=False,
        )

    def test_list_view_status_code(self):
        """Verifica que la lista de vehículos cargue correctamente."""
        response = self.client.get(reverse("vehicles:list"))
        self.assertEqual(response.status_code, 200)
        self.assertContains(response, "Corolla")

    def test_list_view_filters(self):
        """Verifica que los filtros funcionen correctamente."""
        # Filtro por precio mínimo
        response = self.client.get(reverse("vehicles:list"), {"min_price": "15000"})
        self.assertContains(response, "Corolla")

        response = self.client.get(reverse("vehicles:list"), {"min_price": "25000"})
        self.assertNotContains(response, "Corolla")

        # Filtro por búsqueda
        response = self.client.get(reverse("vehicles:list"), {"q": "Toyota"})
        self.assertContains(response, "Corolla")

        response = self.client.get(reverse("vehicles:list"), {"q": "Honda"})
        self.assertNotContains(response, "Corolla")


class VehicleCreateViewTest(TestCase):
    """Pruebas para VehicleCreateView."""

    def setUp(self):
        self.client = Client()
        self.brand = Brand.objects.create(name="Toyota")
        self.seller = User.objects.create_user(
            username="seller",
            email="seller@test.com",
            password="testpass123",
            role="admin",
        )
        self.customer = User.objects.create_user(
            username="customer",
            email="customer@test.com",
            password="testpass123",
            role="customer",
        )

    def test_create_vehicle_seller(self):
        """Verifica que un vendedor pueda crear un vehículo."""
        self.client.force_login(self.seller)
        response = self.client.post(
            reverse("vehicles:create"),
            {
                "brand": self.brand.id,
                "model": "Camry",
                "year": 2021,
                "price": 25000.00,
                "mileage": 10000,
                "transmission": "automatic",
                "fuel_type": "gasoline",
                "status": "used",
                "description": "Excelente estado",
                "is_active": True,
            },
        )
        self.assertEqual(response.status_code, 302)  # Redirección
        self.assertEqual(Vehicle.objects.count(), 1)
        self.assertEqual(Vehicle.objects.first().created_by, self.seller)

    def test_create_vehicle_customer_forbidden(self):
        """Verifica que un cliente NO pueda crear un vehículo."""
        self.client.force_login(self.customer)
        response = self.client.get(reverse("vehicles:create"))
        self.assertEqual(response.status_code, 403)  # Forbidden

    def test_create_vehicle_unauthenticated(self):
        """Verifica que usuarios no autenticados no puedan crear vehículos."""

class VehicleUpdateViewTest(TestCase):
    """Pruebas para VehicleUpdateView."""

    def setUp(self):
        self.client = Client()
        self.brand = Brand.objects.create(name="Toyota")
        self.owner = User.objects.create_user(
            username="owner",
            email="owner@test.com",
            password="testpass123",
            role="admin",
        )

        self.other_user = User.objects.create_user(
            username="other",
            email="other@test.com",
            password="testpass123",
            role="admin",
        )

        self.admin_user = User.objects.create_superuser(
            username="admin", email="admin@test.com", password="adminpass123"
        )
        self.vehicle = Vehicle.objects.create(
            brand=self.brand,
            model="Corolla",
            year=2020,
            price=20000.00,
            mileage=15000,
            transmission="automatic",
            fuel_type="gasoline",
            status="used",
            created_by=self.owner,
            is_active=True,
            sold=False,
        )

    def test_update_vehicle_owner(self):
        """Verifica que el propietario pueda actualizar su vehículo."""
        self.client.force_login(self.owner)
        response = self.client.post(
            reverse("vehicles:update", args=[self.vehicle.id]),
            {
                "brand": self.brand.id,
                "model": "Camry",
                "year": 2021,
                "price": 26000.00,
                "mileage": 12000,
                "transmission": "automatic",
                "fuel_type": "gasoline",
                "status": "used",
                "description": "Nuevo estado",
                "is_active": True,
            },
        )
        self.assertEqual(response.status_code, 302)
        self.vehicle.refresh_from_db()
        self.assertEqual(self.vehicle.model, "Camry")
        self.assertEqual(self.vehicle.price, 26000.00)

    def test_update_vehicle_other_user_forbidden(self):
        """Verifica que otro usuario NO pueda actualizar el vehículo (a menos que sea admin)."""
        self.client.force_login(self.other_user)  # other_user es admin
        response = self.client.get(reverse("vehicles:update", args=[self.vehicle.id]))
        self.assertEqual(response.status_code, 200)  # Admin puede ver el formulario

    def test_update_vehicle_admin_allowed(self):
        """Verifica que un administrador pueda actualizar cualquier vehículo."""
        self.client.force_login(self.admin_user)
        response = self.client.post(
            reverse("vehicles:update", args=[self.vehicle.id]),
            {
                "brand": self.brand.id,
                "model": "Prius",
                "year": 2022,
                "price": 30000.00,
                "mileage": 5000,
                "transmission": "automatic",
                "fuel_type": "hybrid",
                "status": "new",
                "description": "Vehiculo híbrido",
                "is_active": True,
            },
        )
        self.assertEqual(response.status_code, 302)
        self.vehicle.refresh_from_db()
        self.assertEqual(self.vehicle.model, "Prius")

class Vehicle(models.Model):
    TRANSMISSION_CHOICES = (
        ("manual", "Manual"),
        ("automatic", "Automática"),
        ("cvt", "CVT"),
    )
    FUEL_CHOICES = (
        ("gasoline", "Gasolina"),
        ("diesel", "Diésel"),
        ("electric", "Eléctrico"),
        ("hybrid", "Híbrido"),
    )
    STATUS_CHOICES = (
        ("new", "Nuevo"),
        ("used", "Usado"),
    )

    brand = models.ForeignKey(Brand, on_delete=models.CASCADE, related_name="vehicles")
    model = models.CharField(max_length=100)
    year = models.PositiveIntegerField()
    price = models.DecimalField(max_digits=12, decimal_places=2)
    mileage = models.PositiveIntegerField(help_text="Kilometraje (km)")
    transmission = models.CharField(max_length=20, choices=TRANSMISSION_CHOICES)
    fuel_type = models.CharField(max_length=20, choices=FUEL_CHOICES)
    status = models.CharField(max_length=10, choices=STATUS_CHOICES, default="used")
    description = models.TextField(blank=True)
    main_image = models.ImageField(
        upload_to="vehicles/",
        blank=True,
        null=True,
        validators=[FileExtensionValidator(["jpg", "jpeg", "png", "gif"])],
    )
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    created_by = models.ForeignKey(
        User, on_delete=models.SET_NULL, null=True, related_name="vehicles_created"
    )
    is_active = models.BooleanField(default=True)
    sold = models.BooleanField(default=False)
    reserved = models.BooleanField(default=False)
    reserved_by = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.SET_NULL,
        null=True,
        blank=True,
        related_name="reserved_vehicles",
    )
    view_count = models.PositiveIntegerField(default=0)

    def __str__(self):
        return f"{self.brand.name} {self.model} ({self.year})"

    def mark_as_sold(self, buyer):
        """Marca el vehículo como vendido (con transacción)."""
        from sales.models import Sale

        with transaction.atomic():
            self.sold = True
            self.is_active = False
            self.save()
            return Sale.objects.create(
                vehicle=self,
                buyer=buyer,
                seller=self.created_by,
                amount=self.price,
                status="completed",
            )

class VehicleListView(ListView):
    model = Vehicle
    template_name = "vehicles/list.html"
    context_object_name = "vehicles"
    paginate_by = 12
    ordering = ["-created_at"]

    def get_queryset(self):
        qs = (
            super()
            .get_queryset()
            .select_related("brand", "created_by")
            .filter(is_active=True)
        )

        # Filtros con validación
        brand = self.request.GET.get("brand")
        if brand:
            qs = qs.filter(brand__name__icontains=brand)

        # Validar y sanitizar precios
        min_price = self.request.GET.get("min_price")
        if min_price and min_price.isdigit():
            qs = qs.filter(price__gte=int(min_price))

        max_price = self.request.GET.get("max_price")
        if max_price and max_price.isdigit():
            qs = qs.filter(price__lte=int(max_price))

        # Búsqueda por texto (opcional)
        q = self.request.GET.get("q")
        if q:
            qs = qs.filter(
                Q(model__icontains=q)
                | Q(brand__name__icontains=q)
                | Q(description__icontains=q)
            )

        return qs


class VehicleDetailView(TrackVehicleViewMixin, DetailView):
    model = Vehicle
    template_name = "vehicles/detail.html"

    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        # Verificar si el usuario puede editar este vehículo
        vehicle = self.get_object()
        user = self.request.user

        pending_offer = Sale.objects.filter(
            vehicle=vehicle, status="offer_pending"
        ).first()
        context["pending_offer"] = pending_offer

        context["can_edit"] = user.is_authenticated and (
            user.role == "admin" or user.is_superuser or vehicle.created_by == user
        )
        return context

class ProfileForm(forms.ModelForm):
    phone = forms.CharField(max_length=20, required=False)
    address = forms.CharField(widget=forms.Textarea, required=False)
    payment_email = forms.EmailField(required=False, validators=[validate_email_domain])
    bank_account = forms.CharField(max_length=100, required=False)
    payment_phone = forms.CharField(max_length=20, required=False)

    class Meta:
        model = User
        fields = ("username", "email", "first_name", "last_name")

    def clean_email(self):
        email = self.cleaned_data.get("email")
        if not verify_email_api(email):
            raise ValidationError("El correo electrónico no parece ser válido.")
        if email:
            email = email.lower()
            # Excluir al propio usuario de la consulta (cuando se edita)
            if self.instance and self.instance.pk:
                if (
                    User.objects.filter(email__iexact=email)
                    .exclude(pk=self.instance.pk)
                    .exists()
                ):
                    raise ValidationError(
                        "Este correo electrónico ya está registrado por otro usuario."
                    )
            else:
                # Caso de creación (aunque no se usa aquí)
                if User.objects.filter(email__iexact=email).exists():
                    raise ValidationError("Este correo electrónico ya está registrado.")
        return email

    def save(self, commit=True):
        user = super().save(commit=commit)
        profile, created = Profile.objects.get_or_create(user=user)
        profile.phone = self.cleaned_data["phone"]
        profile.address = self.cleaned_data["address"]
        profile.payment_email = self.cleaned_data["payment_email"]
        profile.bank_account = self.cleaned_data["bank_account"]
        profile.payment_phone = self.cleaned_data["payment_phone"]
        if commit:
            profile.save()
        return user

import re

import dns.resolver
from django.core.exceptions import ValidationError


def validate_email_domain(value):
    try:
        domain = value.split("@")[1]
        dns.resolver.resolve(domain, "MX")
    except (IndexError, dns.resolver.NoAnswer, dns.resolver.NXDOMAIN):
        raise ValidationError(
            "El dominio del correo electrónico no es válido o no tiene servidores de correo."
        )
    except dns.resolver.NoNameservers:
        raise ValidationError("El dominio del correo electrónico no es válido.")

import uuid

from django.conf import settings
from django.contrib.auth.models import AbstractUser
from django.db import models
from django.db.models.signals import post_save
from django.dispatch import receiver
from django.utils import timezone

token = models.UUIDField(default=uuid.uuid4, editable=False, unique=True)
expires_at = models.DateTimeField()


class User(AbstractUser):
    email = models.EmailField(unique=True)
    ROLE_CHOICES = (
        ("admin", "Administrador"),
        ("seller", "Comerciante"),
        ("customer", "Cliente"),
    )
    role = models.CharField(max_length=20, choices=ROLE_CHOICES, default="customer")

    # No definas groups ni user_permissions aquí, ya vienen de AbstractUser

    def __str__(self):
        return self.username


class Profile(models.Model):
    user = models.OneToOneField(User, on_delete=models.CASCADE, related_name="profile")
    phone = models.CharField(max_length=20, blank=True)
    address = models.TextField(blank=True)
    # Datos de pago para comerciantes
    payment_email = models.EmailField(
        blank=True,
        null=True,
        help_text="Email para recibir pagos (ej. PayPal, MercadoPago)",
    )
    bank_account = models.CharField(
        max_length=100, blank=True, null=True, help_text="Número de cuenta bancaria"
    )
    payment_phone = models.CharField(
        max_length=20,
        blank=True,
        null=True,
        help_text="Teléfono para pagos (ej. Bizum, WhatsApp)",
    )

    def __str__(self):
        return f"Perfil de {self.user.username}"


@receiver(post_save, sender=User)
def create_user_profile(sender, instance, created, **kwargs):
    if created:
        Profile.objects.create(user=instance)


def get_expiration_time():
    return timezone.now() + timedelta(hours=24)


class EmailVerification(models.Model):
    user = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.CASCADE,
        related_name="email_verifications",
    )
    email = models.EmailField()
    token = models.UUIDField(
        default=uuid.uuid4, editable=False, unique=True
    )  # <-- Agregar
    expires_at = models.DateTimeField(default=get_expiration_time)
    created_at = models.DateTimeField(auto_now_add=True)
    is_verified = models.BooleanField(default=False)

    def __str__(self):
        return f"Verificación de {self.user.email}"

def verify_email_api(email):
    """
    Verifica la validez de un email usando una API externa.
    """
    api_key = getattr(settings, "EMAIL_VERIFICATION_API_KEY", None)
    if not api_key:
        # Si no hay API key, omitimos la verificación (no bloquear)
        return True
    url = f"https://api.hunter.io/v2/email-verifier?email={email}&api_key={api_key}"
    try:
        response = requests.get(url, timeout=5)
        data = response.json()
        if data.get("data", {}).get("status") == "valid":
            return True
        return False
    except Exception:
        # Si falla la API, permitimos el registro (no bloquear)
        return True

def verify_email(request, token):
    try:
        verification = EmailVerification.objects.get(token=token, is_verified=False)
        if verification.is_expired():
            messages.error(
                request,
                "El enlace ha expirado. Solicita un nuevo enlace de verificación.",
            )
            return redirect("users:login")
        verification.is_verified = True
        verification.save()
        # Opcional: marcar el email del usuario como verificado (si agregas un campo `email_verified` en User)
        messages.success(
            request, "Email verificado correctamente. ¡Ya puedes iniciar sesión!"
        )
    except EmailVerification.DoesNotExist:
        messages.error(request, "Enlace inválido o ya utilizado.")
    return redirect("users:login")

from django.conf import settings
from django.db import models

from vehicles.models import Vehicle


class Sale(models.Model):
    STATUS_CHOICES = (
        ("offer_pending", "Oferta pendiente"),
        ("offer_accepted", "Oferta aceptada"),
        ("offer_rejected", "Oferta rechazada"),
        ("completed", "Completada"),
        ("cancelled", "Cancelada"),
    )
    vehicle = models.ForeignKey(Vehicle, on_delete=models.CASCADE, related_name="sales")
    buyer = models.ForeignKey(
        settings.AUTH_USER_MODEL, on_delete=models.CASCADE, related_name="purchases"
    )
    seller = models.ForeignKey(
        settings.AUTH_USER_MODEL, on_delete=models.CASCADE, related_name="sales"
    )
    amount = models.DecimalField(
        max_digits=12, decimal_places=2
    )  # Precio original del vehículo
    offered_price = models.DecimalField(
        max_digits=12, decimal_places=2, null=True, blank=True
    )  # Precio ofertado
    status = models.CharField(
        max_length=20, choices=STATUS_CHOICES, default="offer_pending"
    )
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    buyer_phone = models.CharField(max_length=20, blank=True)
    buyer_email = models.EmailField(blank=True)
    notes = models.TextField(blank=True)

    def __str__(self):
        return f"Venta de {self.vehicle} - {self.buyer.username}"

    class Meta:
        ordering = ["-created_at"]

def send_email(subject, template_name, context, to, from_email=None):
    """
    Función auxiliar para enviar emails con plantillas HTML y texto plano.
    """
    try:
        # Renderizar HTML
        html_message = render_to_string(template_name, context)
        # Crear versión texto plano (para clientes que no soportan HTML)
        plain_message = strip_tags(html_message)

        send_mail(
            subject=subject,
            message=plain_message,
            from_email=from_email or settings.DEFAULT_FROM_EMAIL,
            recipient_list=to,
            html_message=html_message,
            fail_silently=False,
        )
        logger.info(f"Email enviado a {', '.join(to)}: {subject}")
    except Exception as e:
        logger.error(f"Error al enviar email a {', '.join(to)}: {str(e)}")
        # No lanzamos excepción para no interrumpir la compra

@login_required
@transaction.atomic
def reserve_vehicle(request, vehicle_id):
    vehicle = get_object_or_404(
        Vehicle.objects.select_for_update(),
        id=vehicle_id,
        is_active=True,
        sold=False,
        reserved=False,
    )

    if vehicle.created_by == request.user:
        messages.error(request, "No puedes reservar tu propio vehículo.")
        return redirect("vehicles:detail", pk=vehicle_id)

    # Crear la venta
    sale = Sale.objects.create(
        vehicle=vehicle,
        buyer=request.user,
        seller=vehicle.created_by,
        amount=vehicle.price,
        status="reserved",
        buyer_phone=(
            request.user.profile.phone if hasattr(request.user, "profile") else ""
        ),
        buyer_email=request.user.email,
    )

    # Marcar vehículo como reservado
    vehicle.reserved = True
    vehicle.reserved_by = request.user
    vehicle.save()

    # Obtener administrador
    admin_user = get_admin_user()

    # Enviar notificaciones
    try:
        send_reservation_notifications(
            vehicle=vehicle, buyer=request.user, admin_user=admin_user, sale=sale
        )
    except Exception as e:
        logger.error(f"Error al enviar email de reserva: {e}")
        # No interrumpir el flujo

    messages.success(
        request,
        f"¡Vehículo reservado! {vehicle.brand.name} {vehicle.model} ha sido reservado.",
    )
    return redirect("vehicles:detail", pk=vehicle_id)

def cancel_reservation(request, vehicle_id):
    vehicle = get_object_or_404(
        Vehicle.objects.select_for_update(),
        id=vehicle_id,
        reserved=True,
        reserved_by=request.user,
    )

    # Obtener la venta asociada
    sale = Sale.objects.filter(
        vehicle=vehicle, buyer=request.user, status="reserved"
    ).first()
    if sale:
        sale.status = "cancelled"
        sale.save()

    # Liberar vehículo
    vehicle.reserved = False
    vehicle.reserved_by = None
    vehicle.save()

    # Obtener administrador
    admin_user = get_admin_user()

    # Enviar email de cancelación al administrador
    if admin_user:
        try:
            send_admin_cancellation_notification(vehicle, request.user, admin_user)
        except Exception as e:
            logger.error(f"Error al enviar email de cancelación al admin: {e}")
    # Enviar email de cancelación al comprador
    try:
        send_cancellation_notification(vehicle, request.user)
    except Exception as e:
        logger.error(f"Error al enviar email de cancelación al comprador: {e}")

    messages.info(
        request, f"Has cancelado la reserva de {vehicle.brand.name} {vehicle.model}."
    )
    return redirect("vehicles:detail", pk=vehicle_id)

@login_required
@transaction.atomic
def make_offer(request, vehicle_id):
    vehicle = get_object_or_404(
        Vehicle.objects.select_for_update(),
        id=vehicle_id,
        is_active=True,
        sold=False,
    )

    if vehicle.created_by == request.user:
        messages.error(request, "No puedes hacer una oferta en tu propio vehículo.")
        return redirect("vehicles:detail", pk=vehicle_id)

    # Verificar si ya hay una oferta pendiente (cualquier usuario)
    pending_offer = Sale.objects.filter(vehicle=vehicle, status="offer_pending").first()
    if pending_offer:
        messages.error(request, "Ya hay una oferta pendiente para este vehículo.")
        return redirect("vehicles:detail", pk=vehicle_id)

    # Obtener la última oferta del usuario para este vehículo (cualquier estado)
    last_offer = (
        Sale.objects.filter(vehicle=vehicle, buyer=request.user)
        .order_by("-created_at")
        .first()
    )

    if request.method == "POST":
        offered_price = request.POST.get("offered_price")
        if not offered_price:
            messages.error(request, "Debes ingresar un precio.")
            return redirect("vehicles:detail", pk=vehicle_id)

        try:
            offered_price = Decimal(offered_price)
            if offered_price <= 0:
                raise ValueError
            if offered_price >= vehicle.price:
                messages.error(
                    request, "El precio ofertado debe ser menor al precio publicado."
                )
                return redirect("vehicles:detail", pk=vehicle_id)

            # Validar que no sea mayor que la última oferta del usuario (si existe)
            if last_offer and offered_price > last_offer.offered_price:
                messages.error(
                    request,
                    f"Tu nueva oferta no puede ser mayor que tu última oferta de ${last_offer.offered_price}. Debes ofertar un precio menor.",
                )
                return redirect("vehicles:detail", pk=vehicle_id)

        except (ValueError, TypeError):
            messages.error(request, "Ingresa un precio válido.")
            return redirect("vehicles:detail", pk=vehicle_id)

        # Crear la oferta
        sale = Sale.objects.create(
            vehicle=vehicle,
            buyer=request.user,
            seller=vehicle.created_by,
            amount=vehicle.price,
            offered_price=offered_price,
            status="offer_pending",
            buyer_phone=(
                request.user.profile.phone if hasattr(request.user, "profile") else ""
            ),
            buyer_email=request.user.email,
        )

        # Obtener administrador
        admin_user = get_admin_user()

        # Enviar notificaciones
        try:
            send_offer_notifications(vehicle, request.user, admin_user, sale)
        except Exception as e:
            logger.error(f"Error al enviar email de oferta: {e}")

        messages.success(
            request,
            f"¡Oferta enviada! {vehicle.brand.name} {vehicle.model} - Tu oferta de ${offered_price} está en revisión.",
        )
        return redirect("vehicles:detail", pk=vehicle_id)

    return redirect("vehicles:detail", pk=vehicle_id)
