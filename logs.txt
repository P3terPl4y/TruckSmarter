def main():

from django.conf import settings
from django.db import models

from vehicles.models import Vehicle


class Sale(models.Model):
    STATUS_CHOICES = (
        ("pending", "Pendiente"),
        ("completed", "Completada"),
        ("cancelled", "Cancelada"),
    )
    vehicle = models.ForeignKey(Vehicle, on_delete=models.CASCADE, related_name="sales")
    buyer = models.ForeignKey(
        settings.AUTH_USER_MODEL, on_delete=models.CASCADE, related_name="purchases"
    )
    seller = models.ForeignKey(
        settings.AUTH_USER_MODEL, on_delete=models.CASCADE, related_name="sales"
    )
    amount = models.DecimalField(max_digits=12, decimal_places=2)
    status = models.CharField(max_length=20, choices=STATUS_CHOICES, default="pending")
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    # Datos de pago del comprador (opcional)
    buyer_phone = models.CharField(max_length=20, blank=True)
    buyer_email = models.EmailField(blank=True)
    notes = models.TextField(blank=True)

    def __str__(self):
        return f"Venta de {self.vehicle} - {self.buyer.username}"

    class Meta:
        ordering = ["-created_at"]

def send_purchase_notifications(vehicle, buyer, seller, sale):
    """
    Envía emails de confirmación al comprador y al vendedor después de una compra.
    
    Args:
        vehicle: Objeto Vehicle vendido
        buyer: Objeto User comprador
        seller: Objeto User vendedor
        sale: Objeto Sale creado
    """
    
    # Datos del vehículo
    vehicle_details = {
        'brand': vehicle.brand.name,
        'model': vehicle.model,
        'year': vehicle.year,
        'price': vehicle.price,
        'description': vehicle.description[:200],  # Resumen
    }
    
    # Datos del comprador (para el vendedor)
    buyer_details = {
        'full_name': buyer.get_full_name() or buyer.username,
        'email': buyer.email,
        'phone': buyer.profile.phone if hasattr(buyer, 'profile') else 'No especificado',
        'address': buyer.profile.address if hasattr(buyer, 'profile') else 'No especificada',
    }
    
    # Datos del vendedor (para el comprador)
    seller_details = {
        'full_name': seller.get_full_name() or seller.username,
        'email': seller.email,
        'phone': seller.profile.phone if hasattr(seller, 'profile') else 'No especificado',
        'address': seller.profile.address if hasattr(seller, 'profile') else 'No especificada',
    }
    
    context = {
        'vehicle': vehicle_details,
        'buyer': buyer_details,
        'seller': seller_details,
        'sale_id': sale.id,
        'sale_amount': sale.amount,
        'sale_date': sale.created_at,
        'site_name': 'TruckSmarter',
        'site_url': 'http://localhost:8000',  # En producción, usar variable de entorno
    }
    
    # 1. Email para el comprador (confirmación)
    send_email(
        subject=f'¡Compra confirmada! Has comprado {vehicle_details["brand"]} {vehicle_details["model"]}',
        template_name='emails/purchase_confirmation.html',
        context=context,
        to=[buyer.email],
        from_email=settings.DEFAULT_FROM_EMAIL,
    )
    
    # 2. Email para el vendedor (notificación de venta)
    send_email(
        subject=f'¡Tu vehículo ha sido vendido! {vehicle_details["brand"]} {vehicle_details["model"]}',
        template_name='emails/sale_notification.html',
        context=context,
        to=[seller.email],
        from_email=settings.DEFAULT_FROM_EMAIL,
    )

def send_email(subject, template_name, context, to, from_email=None):
    """
    Función auxiliar para enviar emails con plantillas HTML y texto plano.
    """
    try:
        # Renderizar HTML
        html_message = render_to_string(template_name, context)
        # Crear versión texto plano (para clientes que no soportan HTML)
        plain_message = strip_tags(html_message)
        
        send_mail(
            subject=subject,
            message=plain_message,
            from_email=from_email or settings.DEFAULT_FROM_EMAIL,
            recipient_list=to,
            html_message=html_message,
            fail_silently=False,
        )
        logger.info(f"Email enviado a {', '.join(to)}: {subject}")

from django.shortcuts import render, get_object_or_404, redirect
from django.contrib.auth.decorators import login_required
from django.contrib import messages
from django.db import transaction
from django.db.models import F
from django.urls import reverse
from vehicles.models import Vehicle
from .models import Sale
from .services import send_purchase_notifications  # <--- Importar


@login_required
@transaction.atomic
def purchase_vehicle(request, vehicle_id):
    """
    Procesa la compra de un vehículo con transacción atómica y envío de emails.
    """
    # Bloquear la fila para evitar compras concurrentes
    vehicle = get_object_or_404(
        Vehicle.objects.select_for_update(),
        id=vehicle_id,
        is_active=True,
        sold=False
    )

    # No permitir comprar propio vehículo
    if vehicle.created_by == request.user:
        messages.error(request, "No puedes comprar tu propio vehículo.")
        return redirect("vehicles:detail", pk=vehicle_id)

    # Crear la venta
    sale = Sale.objects.create(
        vehicle=vehicle,
        buyer=request.user,
        seller=vehicle.created_by,
        amount=vehicle.price,
        status="completed",
        buyer_phone=request.user.profile.phone if hasattr(request.user, "profile") else "",
        buyer_email=request.user.email,
    )

    # Marcar el vehículo como vendido
    vehicle.sold = True
    vehicle.is_active = False
    vehicle.save()

    # --- ENVIAR NOTIFICACIONES POR EMAIL ---
    try:
        send_purchase_notifications(
            vehicle=vehicle,
            buyer=request.user,
            seller=vehicle.created_by,
            sale=sale
        )
    except Exception as e:
        # El email falla, pero no debemos interrumpir la compra
        # Logueamos el error (ya se loguea dentro de la función)
        pass

    messages.success(
        request,
        f"¡Felicidades! Has comprado {vehicle.brand.name} {vehicle.model} por ${vehicle.price}. "
        "Hemos enviado un email de confirmación a tu correo."
    )

    return redirect("vehicles:detail", pk=vehicle_id)
@login_required
def purchase_history(request):
    """Ver historial de compras del usuario."""
    sales = Sale.objects.filter(buyer=request.user).select_related("vehicle", "seller")
    return render(request, "sales/history.html", {"sales": sales})

    )

    sold = models.BooleanField(default=False)

            self.save()

                buyer=buyer,
                seller=self.created_by,
                amount=self.price,
                status="completed",
            )

class VehicleListView(ListView):
    model = Vehicle
    template_name = "vehicles/list.html"
    context_object_name = "vehicles"
    paginate_by = 12
    ordering = ["-created_at"]

    def get_queryset(self):
        qs = (
            super()
            .get_queryset()
            .select_related("brand", "created_by")
            .filter(is_active=True)
        )

        # Filtros con validación
        brand = self.request.GET.get("brand")
        if brand:
            qs = qs.filter(brand__name__icontains=brand)

        # Validar y sanitizar precios
        min_price = self.request.GET.get("min_price")
        if min_price and min_price.isdigit():
            qs = qs.filter(price__gte=int(min_price))

        max_price = self.request.GET.get("max_price")
        if max_price and max_price.isdigit():
            qs = qs.filter(price__lte=int(max_price))

        # Búsqueda por texto (opcional)
        q = self.request.GET.get("q")
        if q:
            qs = qs.filter(
                Q(model__icontains=q)
                | Q(brand__name__icontains=q)
                | Q(description__icontains=q)
            )

        return qs

class ProfileView(LoginRequiredMixin, UpdateView):
    model = User
    form_class = ProfileForm
    template_name = "users/profile.html"
    success_url = reverse_lazy("users:profile")

    def get_object(self):
        return self.request.user

    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        # Crear perfil si no existe
        profile, created = Profile.objects.get_or_create(user=self.request.user)
        context["profile"] = profile

        # Mostrar vehículos si es comerciante o admin
        if self.request.user.role in ["seller", "admin"]:
            from vehicles.models import Vehicle

            context["vehicles"] = Vehicle.objects.filter(
                created_by=self.request.user, is_active=True
            ).select_related("brand")

        return context

    def form_valid(self, form):
        messages.success(self.request, "¡Perfil actualizado correctamente!")
        return super().form_valid(form)

    def form_invalid(self, form):
        messages.error(
            self.request, "Error al actualizar el perfil. Revisa los campos."
        )
        return super().form_invalid(form)
